pipeline {
    agent any

    environment {
        IMAGE_NAME = "flask-reg-app"
    }

    stages {

        stage('Clone Repo') {
            steps {
                git branch: 'main', url: 'https://github.com/shajahan1916/flask-labs.git'
            }
        }

        stage('Build Docker Image (Local)') {
            steps {
                bat "docker build -t %IMAGE_NAME%:latest ."
            }
        }

        stage('Deploy to Local Kubernetes') {
            steps {
                bat "kubectl apply -f k8s\\k8s-deploy.yaml"
                bat "kubectl rollout restart deployment flask-deployment"
            }
        }
    }
}
